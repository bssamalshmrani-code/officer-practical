    const PRACTICAL_WEBHOOK = "https://discord.com/api/webhooks/1573071291605328022/RX_RtQ18T2-XA1Z1rsAD6I6ZH7ht1jSsiy7z3zBbjzFhGv5A4gIW-2fidOsByuB-opAK";

    function updateScores() {
      const d = Number(document.getElementById("discipline").value);
      const f = Number(document.getElementById("field").value);
      const g = Number(document.getElementById("general").value);
      const n = Number(document.getElementById("negotiation").value);

      document.getElementById("valDiscipline").textContent = `${d} / 10`;
      document.getElementById("valField").textContent = `${f} / 10`;
      document.getElementById("valGeneral").textContent = `${g} / 10`;
      document.getElementById("valNegotiation").textContent = `${n} / 10`;
      document.getElementById("totalScore").textContent = `${d + f + g + n} / 40`;
    }

    document.getElementById("evalForm").addEventListener("submit", async function (e) {
      e.preventDefault();
      const btn = document.getElementById("submitBtn");
      const alertBox = document.getElementById("statusAlert");

      btn.disabled = true;
      btn.textContent = "جاري الإرسال...";
      alertBox.className = "alert";

      const studentName = document.getElementById("studentName").value.trim();
      const studentCode = document.getElementById("studentCode").value.trim().toUpperCase();
      const evaluatorName = document.getElementById("evaluatorName").value.trim();
      const d = Number(document.getElementById("discipline").value);
      const f = Number(document.getElementById("field").value);
      const g = Number(document.getElementById("general").value);
      const n = Number(document.getElementById("negotiation").value);
      const total = d + f + g + n;
      const notes = document.getElementById("notes").value.trim() || "لا توجد ملاحظات";

      const payload = {
        username: "رصد الاختبار العملي",
        embeds: [
          {
            title: "🛡️ تقرير تقييم الاختبار العملي — بانتظار الاعتماد",
            color: 13938487,
            fields: [
              { name: "👤 اسم الطالب", value: `**${studentName}**`, inline: true },
              { name: "🎖️ الكود العسكري", value: `**${studentCode}**`, inline: true },
              { name: "👮‍♂️ المُقيّم", value: evaluatorName, inline: true },
              {
                name: "📊 تفاصيل التقييم العملي",
                value:
                  `• **قسم الانضباط:** ( ${d} من 10 )\n` +
                  `• **قسم الميدان:** ( ${f} من 10 )\n` +
                  `• **قسم الـعـام:** ( ${g} من 10 )\n` +
                  `• **قسم التفاوض:** ( ${n} من 10 )`,
                inline: false
              },
              { name: "🏆 المجموع العملي", value: `**${total} / 40**`, inline: true },
              { name: "⏳ الحالة", value: "**بانتظار اعتماد المسؤول**", inline: true },
              { name: "📝 ملاحظات اللجنة", value: `\`\`\`${notes}\`\`\``, inline: false }
            ],
            footer: { text: `PRACTICAL_DATA|${studentName}|${studentCode}|${d}|${f}|${g}|${n}` },
            timestamp: new Date().toISOString()
          }
        ]
      };

      try {
        const res = await fetch(PRACTICAL_WEBHOOK, {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify(payload)
        });

        if (res.ok) {
          alertBox.textContent = "✅ تم إرسال تقييم الطالب بنجاح إلى روم الاختبار العملي بانتظار اعتماد المسؤول.";
          alertBox.className = "alert success";
          document.getElementById("evalForm").reset();
          updateScores();
        } else {
          throw new Error("فشل الإرسال للديسكورد");
        }
      } catch (err) {
        alertBox.textContent = "❌ تعذر الإرسال، تأكد من اتصال الإنترنت أو رابط الويب هوك.";
        alertBox.className = "alert error";
      } finally {
        btn.disabled = false;
        btn.textContent = "إرسال التقييم للاعتماد في الديسكورد";
      }
    });
